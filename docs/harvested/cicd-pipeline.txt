# GhostLink CI/CD Pipeline Configuration for Unified Dashboard
# This file defines the automated testing and deployment workflows

# This is a template that can be used with GitHub Actions, GitLab CI, or other CI/CD systems
# The actual implementation may vary based on your specific CI/CD platform

# Version: 1.0.0
# Date: 2025-10-11

name: Unified Dashboard CI/CD Pipeline

# Define when this workflow should run
on:
  # Run on push to main branch
  push:
    branches: [ main ]
  
  # Run on pull requests targeting main
  pull_request:
    branches: [ main ]
  
  # Allow manual triggering
  workflow_dispatch:

# Environment variables used throughout the pipeline
env:
  NODE_VERSION: 18
  CACHE_KEY: dashboard-deps-${{ hashFiles('**/package-lock.json') }}
  GHOSTLINK_DIR: /Users/ghost/GhostLink

# Define the pipeline jobs
jobs:
  # Step 1: Validate and test the code
  validate:
    name: Lint and Test
    runs-on: ubuntu-latest
    
    steps:
      - name: Checkout code
        uses: actions/checkout@v3
      
      - name: Setup Node.js
        uses: actions/setup-node@v3
        with:
          node-version: ${{ env.NODE_VERSION }}
          cache: 'npm'
      
      - name: Install dependencies
        run: npm ci
      
      - name: Lint code
        run: npm run lint
      
      - name: Type check
        run: npx tsc --noEmit
      
      - name: Run unit tests
        run: npm run test
      
      - name: Archive test results
        uses: actions/upload-artifact@v3
        with:
          name: test-results
          path: coverage/
  
  # Step 2: Build the application
  build:
    name: Build Application
    needs: validate
    runs-on: ubuntu-latest
    
    steps:
      - name: Checkout code
        uses: actions/checkout@v3
      
      - name: Setup Node.js
        uses: actions/setup-node@v3
        with:
          node-version: ${{ env.NODE_VERSION }}
          cache: 'npm'
      
      - name: Install dependencies
        run: npm ci
      
      - name: Build application
        run: npm run build
      
      - name: Archive build artifacts
        uses: actions/upload-artifact@v3
        with:
          name: build-artifacts
          path: dist/
  
  # Step 3: Deploy to staging environment
  deploy-staging:
    name: Deploy to Staging
    needs: build
    runs-on: ubuntu-latest
    if: github.event_name == 'push' && github.ref == 'refs/heads/main'
    
    steps:
      - name: Download build artifacts
        uses: actions/download-artifact@v3
        with:
          name: build-artifacts
          path: dist
      
      - name: Deploy to Staging (Vercel)
        # Use Vercel CLI to deploy
        run: |
          npm install -g vercel
          vercel --token ${{ secrets.VERCEL_TOKEN }} --prod
  
  # Step 4: Run GhostLink integration
  ghostlink-integration:
    name: GhostLink Platform Integration
    needs: deploy-staging
    runs-on: self-hosted
    if: github.event_name == 'push' && github.ref == 'refs/heads/main'
    
    steps:
      - name: Run GhostLink platform sync
        run: |
          bash ${{ env.GHOSTLINK_DIR }}/ghostlink_platform.sh sync all
      
      - name: Create GhostLink snapshot
        run: |
          bash ${{ env.GHOSTLINK_DIR }}/ghostlink_platform.sh backup ./
      
      - name: Run GhostLink validation
        run: |
          bash ${{ env.GHOSTLINK_DIR }}/ghostlink_control.sh monitor

# GhostLink Integration Functions
# These are shell functions that can be used by the CI/CD pipeline

functions:
  # Function to check for security vulnerabilities
  security_check: |
    #!/bin/bash
    npm audit --production
    npx snyk test --severity-threshold=high
  
  # Function to generate documentation
  generate_docs: |
    #!/bin/bash
    npx typedoc --entryPointStrategy expand ./src
  
  # Function to notify team about deployment
  notify_team: |
    #!/bin/bash
    curl -X POST \
      -H "Content-Type: application/json" \
      -d '{"text":"New deployment to staging complete"}' \
      $WEBHOOK_URL
  
  # Function to integrate with GhostLink
  ghostlink_sync: |
    #!/bin/bash
    # Sync Google services
    bash "$GHOSTLINK_DIR/ghostlink_services.sh" google drive open
    bash "$GHOSTLINK_DIR/ghostlink_services.sh" google gmail open
    bash "$GHOSTLINK_DIR/ghostlink_services.sh" google calendar open
    
    # Backup project
    bash "$GHOSTLINK_DIR/ghostlink_platform.sh" backup ./
    
    # Validate deployment
    bash "$GHOSTLINK_DIR/ghostlink_control.sh" monitor
