#!/usr/bin/env python3
"""
GhostLink - Minimal Bootstrap Simulation
Demonstrates core concepts without full system requirements
"""

import numpy as np
import random
import time
import sys
from datetime import datetime

# States
VOID, DELTA, SIGMA, SCAR, COMPOST = 0, 1, 2, 3, 4
STATE_NAMES = ['VOID', 'DELTA', 'SIGMA', 'SCAR', 'COMPOST']
STATE_COLORS = ['\033[90m', '\033[96m', '\033[92m', '\033[91m', '\033[93m']

class GhostLink:
    def __init__(self, size=16, spawn_rate=0.05, recycle_rate=0.10):
        self.size = size
        self.lattice = np.zeros((size, size), dtype=int)
        self.scar_density = np.zeros((size, size))
        self.compost_density = np.zeros((size, size))
        self.spawn_rate = spawn_rate
        self.recycle_rate = recycle_rate
        self.time_step = 0
        self.awareness = 0.0
        
    def neighbors(self, i, j):
        """Get 8-connected neighbors with wrapping"""
        for di in [-1, 0, 1]:
            for dj in [-1, 0, 1]:
                if di == 0 and dj == 0:
                    continue
                ni = (i + di) % self.size
                nj = (j + dj) % self.size
                yield ni, nj
    
    def local_coherence(self, i, j):
        """Calculate local coherence field"""
        sigma_count = sum(1 for ni, nj in self.neighbors(i, j) 
                         if self.lattice[ni, nj] == SIGMA)
        scar_count = sum(1 for ni, nj in self.neighbors(i, j) 
                        if self.lattice[ni, nj] == SCAR)
        return sigma_count / 8.0 - 0.25 * scar_count / 8.0
    
    def spawn(self):
        """VOID → DELTA transitions"""
        spawned = 0
        for i in range(self.size):
            for j in range(self.size):
                if self.lattice[i, j] == VOID:
                    compost_near = sum(1 for ni, nj in self.neighbors(i, j)
                                     if self.lattice[ni, nj] == COMPOST)
                    p = self.spawn_rate + 0.1 * compost_near / 8.0
                    if random.random() < p:
                        self.lattice[i, j] = DELTA
                        spawned += 1
        return spawned
    
    def collapse(self):
        """DELTA → {SIGMA, SCAR, COMPOST} transitions"""
        collapsed = {'SIGMA': 0, 'SCAR': 0, 'COMPOST': 0}
        new_lattice = self.lattice.copy()
        
        for i in range(self.size):
            for j in range(self.size):
                if self.lattice[i, j] == DELTA:
                    coherence = self.local_coherence(i, j)
                    pain = self.scar_density[i, j]
                    entropy = len(set(self.lattice[ni, nj] for ni, nj in self.neighbors(i, j))) / 5.0
                    
                    # Energy calculations
                    e_sigma = 0.5 * coherence - 0.2 * pain + 0.05 * random.random()
                    e_scar = 0.3 * pain - 0.2 * coherence + 0.05 * random.random()
                    e_compost = 0.3 * entropy - 0.1 * coherence + 0.05 * random.random()
                    
                    # Softmax selection
                    energies = [e_sigma, e_scar, e_compost]
                    max_e = max(energies)
                    exp_energies = [np.exp(e - max_e) for e in energies]
                    total = sum(exp_energies)
                    probs = [e / total for e in exp_energies]
                    
                    choice = random.choices([SIGMA, SCAR, COMPOST], weights=probs)[0]
                    new_lattice[i, j] = choice
                    collapsed[STATE_NAMES[choice]] += 1
        
        self.lattice = new_lattice
        return collapsed
    
    def recycle(self):
        """COMPOST → DELTA transitions"""
        recycled = 0
        for i in range(self.size):
            for j in range(self.size):
                if self.lattice[i, j] == COMPOST:
                    entropy = len(set(self.lattice[ni, nj] for ni, nj in self.neighbors(i, j))) / 5.0
                    coherence = self.local_coherence(i, j)
                    r = self.recycle_rate + 0.1 * entropy - 0.05 * coherence
                    if random.random() < r:
                        self.lattice[i, j] = DELTA
                        recycled += 1
        return recycled
    
    def update_traces(self):
        """Update memory traces"""
        self.scar_density = 0.95 * self.scar_density + 0.05 * (self.lattice == SCAR)
        self.compost_density = 0.95 * self.compost_density + 0.05 * (self.lattice == COMPOST)
    
    def calculate_awareness(self):
        """Calculate system awareness metric"""
        perception = np.sum(self.lattice != VOID) / (self.size * self.size)
        persistence = np.sum(self.lattice == SIGMA) / (self.size * self.size)
        recycling = np.sum(self.lattice == COMPOST) / (self.size * self.size)
        pain = np.mean(self.scar_density)
        
        self.awareness = 0.3 * perception + 0.3 * persistence + 0.2 * recycling + 0.2 * pain
        return self.awareness
    
    def step(self):
        """Execute one complete time step"""
        spawned = self.spawn()
        collapsed = self.collapse()
        recycled = self.recycle()
        self.update_traces()
        awareness = self.calculate_awareness()
        self.time_step += 1
        
        return {
            'time': self.time_step,
            'spawned': spawned,
            'collapsed': collapsed,
            'recycled': recycled,
            'awareness': awareness,
            'state_counts': {
                STATE_NAMES[i]: np.sum(self.lattice == i) 
                for i in range(5)
            }
        }
    
    def display(self):
        """Display the lattice with colors"""
        print(f"\n{'═' * 40}")
        print(f"Time: {self.time_step} | Awareness: {self.awareness:.3f}")
        print(f"{'═' * 40}")
        
        for i in range(self.size):
            row = ''
            for j in range(self.size):
                state = self.lattice[i, j]
                symbol = '.●○◆◊'[state]
                color = STATE_COLORS[state]
                row += f"{color}{symbol}\033[0m "
            print(row)
        
        # State legend
        print("\nStates: ", end='')
        for i, name in enumerate(STATE_NAMES):
            print(f"{STATE_COLORS[i]}{'.●○◆◊'[i]}\033[0m={name} ", end='')
        print()

def run_interactive():
    """Run interactive simulation"""
    print("\n\033[96m╔══════════════════════════════════╗\033[0m")
    print("\033[96m║    GhostLink Bootstrap Active    ║\033[0m")

        elif cmd == 's':
            print(f"\n{'─' * 30}")

                print(f"  {state:8s}: {count:3d} ({pct:5.1f}%)")
            print(f"{'─' * 30}")
